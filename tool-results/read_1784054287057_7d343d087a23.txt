    66→    66→

   119→   119→            self.name = self.agent.name
   120→   120→        if not self.description:

   129→   129→    def heartbeat(self) -> None:
   130→   130→        """Actualiza last_heartbeat y last_seen."""
   131→   131→        self.last_heartbeat = _now_iso()
   132→   132→        self.last_seen = _now_iso()

   134→   134→    def update_status(self) -> None:
   135→   135→        """Snapshot del estado del agente subyacente."""
   136→   136→        self.status = self.agent.status
   137→   137→        report = self.agent.health() if hasattr(self.agent, "health") else None
   138→   138→        if report is not None:
   139→   139→            self.health_state = report.state

   141→   141→    def record_usage(
   142→   142→        self,
   143→   143→        *,
   144→   144→        success: bool,
   145→   145→        duration_ms: float = 0.0,
   146→   146→    ) -> None:
   147→   147→        """Registra una selección del agente."""
   148→   148→        self.usage_count += 1
   149→   149→        self.total_duration_ms += duration_ms
   150→   150→        self._latencies.append(duration_ms)
   151→   151→        if len(self._latencies) > self.MAX_LATENCIES:
   152→   152→            self._latencies = self._latencies[-self.MAX_LATENCIES:]
   153→   153→        if success:
   154→   154→            self.success_count += 1
   155→   155→        else:
   156→   156→            self.failure_count += 1
   157→   157→            # Sincronizar errores del agente si los tiene.
   158→   158→            last_err = getattr(self.agent, "last_error", None)
   159→   159→            if last_err:
   160→   160→                self.add_error(last_err)
   161→   161→        self.last_seen = _now_iso()
   162→   162→
   163→   163→    def add_error(self, error: str) -> None:
   164→   164→        """Añade un error al historial (con cap)."""
   165→   165→        self.errors.append(error)
   166→   166→        if len(self.errors) > self.MAX_ERRORS:
   167→   167→            self.errors = self.errors[-self.MAX_ERRORS:]
   168→   168→
   169→   169→    @property
   170→   170→    def avg_latency_ms(self) -> float:
   171→   171→        if not self._latencies:
   172→   172→            return 0.0
   173→   173→        return sum(self._latencies) / len(self._latencies)
   174→   174→
   175→   175→    @property
   176→   176→    def success_rate(self) -> float:
   177→   177→        total = self.success_count + self.failure_count
   178→   178→        if total == 0:
   179→   179→            return 1.0
   180→   180→        return self.success_count / total
   181→   181→
   182→   182→    @property
   183→   183→    def uptime_s(self) -> float:
   184→   184→        """Segundos desde created_at (aproximado, basado en epoch)."""
   185→   185→        try:
   186→   186→            created = datetime.fromisoformat(self.created_at).timestamp()
   187→   187→            return _now_epoch() - created
   188→   188→        except (ValueError, OSError):
   189→   189→            return 0.0
   190→   190→
   191→   191→    @property
   192→   192→    def is_healthy(self) -> bool:
   193→   193→        """True si health_state es ALIVE o IDLE."""
   194→   194→        return self.health_state in (HealthState.ALIVE, HealthState.IDLE)
   195→   195→
   196→   196→    @property
   197→   197→    def is_available(self) -> bool:
   198→   198→        """True si está disponible para selección (no DEAD/ERROR/STOPPED)."""
   199→   199→        return (
   200→   200→            self.status in (AgentStatus.IDLE, AgentStatus.CREATED)
   201→   201→            and self.health_state in (HealthState.ALIVE, HealthState.IDLE)
   202→   202→        )

   205→   205→    # Capabilities
   206→   206→    # ------------------------------------------------------------------
   207→   207→
   208→   208→    def provides(self, capability: str) -> bool:
   209→   209→        """True si el agente provee la capability (con wildcard match)."""
   210→   210→        for cap in self.provided_capabilities:
   211→   211→            # Match exacto.
   212→   212→            if cap == capability:
   213→   213→                return True
   214→   214→            # Match wildcard: cap="code.*" matches capability="code.generate".
   215→   215→            if cap.endswith(".*"):
   216→   216→                prefix = cap[:-2]
   217→   217→                if capability == prefix or capability.startswith(prefix + "."):
   218→   218→                    return True
   219→   219→            # Match jerárquico: cap="code" matches capability="code.generate"
   220→   220→            # (el agente que provee "code" puede manejar sub-capabilities).
   221→   221→            if capability.startswith(cap + "."):
   222→   222→                return True
   223→   223→        return False
   224→   224→
   225→   225→    def requires_satisfied(self, available_caps: set[str]) -> bool:
   226→   226→        """True si todas las capabilities requeridas están disponibles."""
   227→   227→        for req in self.required_capabilities:
   228→   228→            if not any(
   229→   229→                req == c or req.startswith(c + ".") or
   230→   230→                (c.endswith(".*") and (req == c[:-2] or req.startswith(c[:-2] + ".")))
   231→   231→                for c in available_caps
   232→   232→            ):
   233→   233→                return False
   234→   234→        return True

   323→   323→    async def register(
   324→   324→        self,
   325→   325→        agent: BaseAgent,
   326→   326→        *,
   327→   327→        provided_capabilities: list[str] | None = None,
   328→   328→        required_capabilities: list[str] | None = None,
   329→   329→        preferred_capabilities: list[str] | None = None,
   330→   330→        priority: int = 0,
   331→   331→        metadata: dict[str, Any] | None = None,
   332→   332→        version: str = "0.1.0",
   333→   333→        replace_existing: bool = True,
   334→   334→    ) -> AgentRecord:
   335→   335→        """Registra un agente.
   336→   336→
   337→   337→        Args:
   338→   338→            agent:                   BaseAgent a registrar.
   339→   339→            provided_capabilities:   Capabilities que provee.
   340→   340→            required_capabilities:   Capabilities que necesita de otros.
   341→   341→            preferred_capabilities:  Capabilities preferidas.
   342→   342→            priority:                Prioridad (mayor = más prioritario).
   343→   343→            metadata:                Metadatos libres.
   344→   344→            version:                 Versión del agente.
   345→   345→            replace_existing:        Si True y ya existe un agente con el
   346→   346→                                     mismo nombre, lo reemplaza.
   347→   347→
   348→   348→        Returns:
   349→   349→            El AgentRecord creado.
   350→   350→
   351→   351→        Raises:
   352→   352→            ValueError: Si ya existe un agente con el mismo nombre y
   353→   353→                replace_existing=False.
   354→   354→            TypeError: Si el objeto no parece un BaseAgent.
   355→   355→        """
   356→   356→        # Duck-typing: verificar atributos clave en lugar de isinstance,
   357→   357→        # para ser robusto frente a re-imports del módulo durante los
   358→   358→        # tests (que crean clases BaseAgent distintas con el mismo nombre).
   359→   359→        if not (
   360→   360→            hasattr(agent, "name")
   361→   361→            and hasattr(agent, "status")
   362→   362→            and hasattr(agent, "health")
   363→   363→            and hasattr(agent, "execute")
   364→   364→            and hasattr(agent, "start")
   365→   365→            and hasattr(agent, "stop")
   366→   366→        ):
   367→   367→            raise TypeError(
   368→   368→                f"Expected BaseAgent (or compatible), got {type(agent).__name__}"
   369→   369→            )
   370→   370→
   371→   371→        async with self._lock:
   372→   372→            existing_id = self._by_name.get(agent.name)
   373→   373→            if existing_id is not None:
   374→   374→                if not replace_existing:
   375→   375→                    raise ValueError(
   376→   376→                        f"Agent '{agent.name}' already registered "
   377→   377→                        f"(id={existing_id})"
   378→   378→                    )
   379→   379→                # Reemplazar.
   380→   380→                old_record = self._records.pop(existing_id, None)
   381→   381→                self._by_name.pop(agent.name, None)
   382→   382→                self._metrics["replaced_total"] += 1
   383→   383→                if old_record is not None:
   384→   384→                    await self._emit("agent.removed", {
   385→   385→                        "agent_id": old_record.id,
   386→   386→                        "name": old_record.name,
   387→   387→                        "reason": "replaced",
   388→   388→                    })
   389→   389→
   390→   390→            record = AgentRecord(
   391→   391→                agent=agent,
   392→   392→                name=agent.name,
   393→   393→                description=agent.description,
   394→   394→                version=version,
   395→   395→                provided_capabilities=list(provided_capabilities or []),
   396→   396→                required_capabilities=list(required_capabilities or []),
   397→   397→                preferred_capabilities=list(preferred_capabilities or []),
   398→   398→                priority=priority,
   399→   399→                metadata=dict(metadata or {}),
   400→   400→            )
   401→   401→            record.update_status()
   402→   402→            record.heartbeat()
   403→   403→            self._records[record.id] = record
   404→   404→            self._by_name[record.name] = record.id
   405→   405→            self._metrics["registered_total"] += 1
   406→   406→
   407→   407→        await self._emit("agent.registered", {
   408→   408→            "agent_id": record.id,
   409→   409→            "name": record.name,
   410→   410→            "version": record.version,
   411→   411→            "provided_capabilities": record.provided_capabilities,
   412→   412→            "priority": record.priority,
   413→   413→        })
   414→   414→        logger.info(
   415→   415→            "Registered agent '%s' (id=%s, caps=%d, priority=%d)",
   416→   416→            record.name, record.id[:8],
   417→   417→            len(record.provided_capabilities), record.priority,
   418→   418→        )
   419→   419→        return record
   420→   420→
   421→   421→    async def unregister(self, id_or_name: str) -> bool:
   422→   422→        """Elimina un agente por id o nombre."""
   423→   423→        async with self._lock:
   424→   424→            record = self._resolve(id_or_name)
   425→   425→            if record is None:
   426→   426→                return False
   427→   427→            self._records.pop(record.id, None)
   428→   428→            self._by_name.pop(record.name, None)
   429→   429→            self._metrics["unregistered_total"] += 1
   430→   430→
   431→   431→        await self._emit("agent.removed", {
   432→   432→            "agent_id": record.id,
   433→   433→            "name": record.name,
   434→   434→            "reason": "unregistered",
   435→   435→        })
   436→   436→        logger.info("Unregistered agent '%s' (id=%s)", record.name, record.id[:8])
   437→   437→        return True
   438→   438→
   439→   439→    async def replace(
   440→   440→        self,
   441→   441→        old_id_or_name: str,
   442→   442→        new_agent: BaseAgent,
   443→   443→        **kwargs: Any,
   444→   444→    ) -> AgentRecord:
   445→   445→        """Reemplaza un agente existente por uno nuevo.
   446→   446→
   447→   447→        Mantiene el mismo record_id si el nombre coincide; si no,
   448→   448→        crea un nuevo registro.
   449→   449→        """
   450→   450→        async with self._lock:
   451→   451→            old_record = self._resolve(old_id_or_name)
   452→   452→            if old_record is None:
   453→   453→                raise KeyError(f"Agent '{old_id_or_name}' not found")
   454→   454→            # Si el nuevo agente tiene el mismo nombre, reusar el id.
   455→   455→            if new_agent.name == old_record.name:
   456→   456→                kwargs.setdefault("priority", old_record.priority)
   457→   457→                kwargs.setdefault("metadata", old_record.metadata)
   458→   458→                kwargs.setdefault("version", old_record.version)
   459→   459→                # Eliminar el viejo (sin emitir evento de removed).
   460→   460→                self._records.pop(old_record.id, None)
   461→   461→                self._by_name.pop(old_record.name, None)
   462→   462→
   463→   463→        # Registrar el nuevo (replace_existing=True para evitar conflictos).
   464→   464→        return await self.register(new_agent, replace_existing=True, **kwargs)
   465→   465→
   466→   466→    # ------------------------------------------------------------------
   467→   467→    # Auto-discovery
   468→   468→    # ------------------------------------------------------------------
   469→   469→
   470→   470→    def add_discovery_hook(self, hook) -> None:
   471→   471→        """Añade un hook de auto-discovery.
   472→   472→
   473→   473→        El hook es un callable async `(registry) -> list[BaseAgent]` que
   474→   474→        devuelve agentes descubiertos. El registry los registrará.
   475→   475→        """
   476→   476→        self._auto_discovery_hooks.append(hook)
   477→   477→
   478→   478→    async def discover(self) -> list[AgentRecord]:
   479→   479→        """Ejecuta todos los hooks de discovery y registra los agentes.
   480→   480→
   481→   481→        Returns:
   482→   482→            Lista de AgentRecord registrados (puede estar vacía).
   483→   483→        """
   484→   484→        discovered: list[AgentRecord] = []
   485→   485→        for hook in self._auto_discovery_hooks:
   486→   486→            try:
   487→   487→                agents = await hook(self)
   488→   488→            except Exception:  # noqa: BLE001
   489→   489→                logger.exception("Discovery hook failed")
   490→   490→                continue
   491→   491→            for agent in agents:
   492→   492→                if not isinstance(agent, BaseAgent):
   493→   493→                    logger.warning(
   494→   494→                        "Discovery hook returned non-BaseAgent: %s",
   495→   495→                        type(agent).__name__,
   496→   496→                    )
   497→   497→                    continue
   498→   498→                # Auto-detect capabilities from agent's objectives/tools.
   499→   499→                caps = self._infer_capabilities(agent)
   500→   500→                try:
   501→   501→                    record = await self.register(
   502→   502→                        agent,
   503→   503→                        provided_capabilities=caps,
   504→   504→                        replace_existing=True,
   505→   505→                    )
   506→   506→                    discovered.append(record)
   507→   507→                    self._metrics["discoveries_total"] += 1
   508→   508→                except Exception:  # noqa: BLE001
   509→   509→                    logger.exception(
   510→   510→                        "Failed to register discovered agent '%s'",
   511→   511→                        agent.name,
   512→   512→                    )
   513→   513→        if discovered:
   514→   514→            logger.info("Discovered and registered %d agents", len(discovered))
   515→   515→        return discovered

   517→   517→    def _infer_capabilities(self, agent: BaseAgent) -> list[str]:

   623→   623→    def find_by_capability(
   624→   624→        self,
   625→   625→        capability: str,
   626→   626→        *,
   627→   627→        only_available: bool = True,
   628→   628→    ) -> list[AgentRecord]:
   629→   629→        """Devuelve agentes que proveen una capability.
   630→   630→
   631→   631→        Args:
   632→   632→            capability:      Nombre o patrón de la capability.
   633→   633→            only_available:  Si True, sólo devuelve agentes disponibles
   634→   634→                             (IDLE/CREATED + healthy).
   635→   635→        """
   636→   636→        result: list[AgentRecord] = []
   637→   637→        for record in self._records.values():
   638→   638→            if only_available and not record.is_available:
   639→   639→                continue
   640→   640→            if record.provides(capability):
   641→   641→                result.append(record)
   642→   642→        # Ordenar por prioridad (desc) y luego por usage_count (asc,
   643→   643→        # para balanceo).
   644→   644→        result.sort(key=lambda r: (-r.priority, r.usage_count))
   645→   645→        return result

   647→   647→    def find_by_capability_pattern(
   648→   648→        self,
   649→   649→        pattern: str,
   650→   650→        *,
   651→   651→        only_available: bool = True,
   652→   652→    ) -> list[AgentRecord]:
   653→   653→        """Como find_by_capability pero con soporte de wildcards.
   654→   654→
   655→   655→        Ej: "code.*" devuelve agentes que proveen code.generate,
   656→   656→        code.review, etc.
   657→   657→        """
   658→   658→        if pattern == "*":
   659→   659→            return [
   660→   660→                r for r in self._records.values()
   661→   661→                if not only_available or r.is_available
   662→   662→            ]
   663→   663→        # Expandir el patrón contra las provided_capabilities de cada agente.
   664→   664→        result: list[AgentRecord] = []
   665→   665→        for record in self._records.values():
   666→   666→            if only_available and not record.is_available:
   667→   667→                continue
   668→   668→            for cap in record.provided_capabilities:
   669→   669→                cap_obj = Capability(name=cap)
   670→   670→                if cap_obj.matches(pattern) or _pattern_matches_cap(pattern, cap):
   671→   671→                    result.append(record)
   672→   672→                    break
   673→   673→        result.sort(key=lambda r: (-r.priority, r.usage_count))
   674→   674→        return result

   688→   688→    def update_health(self) -> int:
   689→   689→        """Actualiza el estado de salud de todos los agentes.
   690→   690→
   691→   691→        Returns el número de agentes cuyo health_state cambió.
   692→   692→        """
   693→   693→        changes = 0
   694→   694→        now_epoch = _now_epoch()
   695→   695→        for record in self._records.values():
   696→   696→            old_state = record.health_state
   697→   697→            record.update_status()
   698→   698→            # Si no hay heartbeat reciente, marcar como DEAD.
   699→   699→            try:
   700→   700→                last_hb = datetime.fromisoformat(record.last_heartbeat).timestamp()
   701→   701→                if now_epoch - last_hb > self._heartbeat_timeout:
   702→   702→                    if record.health_state != HealthState.DEAD:
   703→   703→                        record.health_state = HealthState.DEAD
   704→   704→            except (ValueError, OSError):
   705→   705→                pass
   706→   706→            if record.health_state != old_state:
   707→   707→                changes += 1
   708→   708→                # Emitir evento de cambio de health (best-effort).
   709→   709→                if self._bus is not None:
   710→   710→                    try:
   711→   711→                        asyncio.create_task(self._emit(
   712→   712→                            "agent.health.changed",
   713→   713→                            {
   714→   714→                                "agent_id": record.id,
   715→   715→                                "name": record.name,
   716→   716→                                "old_state": old_state.value,
   717→   717→                                "new_state": record.health_state.value,
   718→   718→                            },
   719→   719→                        ))
   720→   720→                    except Exception:  # noqa: BLE001
   721→   721→                        pass
   722→   722→        self._metrics["heartbeat_checks"] += 1
   723→   723→        self._metrics["health_changes"] += changes
   724→   724→        return changes

   726→   726→    def health(self) -> dict[str, Any]:
   727→   727→        """Health check de todos los agentes."""
   728→   728→        records = self.get_all()
   729→   729→        summary: dict[str, int] = {s.value: 0 for s in HealthState}
   730→   730→        for r in records:
   731→   731→            summary[r.health_state.value] += 1
   732→   732→        return {
   733→   733→            "total": len(records),
   734→   734→            "summary": summary,
   735→   735→            "healthy": summary.get(HealthState.ALIVE.value, 0)
   736→   736→                       + summary.get(HealthState.IDLE.value, 0),
   737→   737→            "degraded": summary.get(HealthState.BUSY.value, 0)
   738→   738→                        + summary.get(HealthState.ERROR.value, 0),
   739→   739→            "dead": summary.get(HealthState.DEAD.value, 0),
   740→   740→            "agents": {r.name: r.health_state.value for r in records},
   741→   741→        }

   743→   743→    async def start_heartbeat_loop(self, interval: float = 10.0) -> None:
   744→   744→        """Arranca un background task que actualiza health periódicamente."""
   745→   745→        if self._heartbeat_task is not None:
   746→   746→            return
   747→   747→
   748→   748→        async def _loop() -> None:
   749→   749→            while True:
   750→   750→                try:
   751→   751→                    self.update_health()
   752→   752→                except Exception:  # noqa: BLE001
   753→   753→                    logger.exception("Heartbeat loop failed")
   754→   754→                await asyncio.sleep(interval)
   755→   755→
   756→   756→        self._heartbeat_task = asyncio.create_task(_loop())
   757→   757→        logger.info("AgentRegistry heartbeat loop started (interval=%.1fs)", interval)
   758→   758→
   759→   759→    async def stop_heartbeat_loop(self) -> None:
   760→   760→        """Detiene el heartbeat loop."""
   761→   761→        if self._heartbeat_task is not None:
   762→   762→            self._heartbeat_task.cancel()
   763→   763→            try:
   764→   764→                await self._heartbeat_task
   765→   765→            except asyncio.CancelledError:
   766→   766→                pass
   767→   767→            self._heartbeat_task = None
   768→   768→            logger.info("AgentRegistry heartbeat loop stopped")

   942→   942→def _pattern_matches_cap(pattern: str, cap_name: str) -> bool:
   943→   943→    """True si `pattern` (con wildcards) matchea `cap_name`."""
   944→   944→    if pattern == "*":
   945→   945→        return True
   946→   946→    if pattern == cap_name:
   947→   947→        return True
   948→   948→    if pattern.endswith(".*"):
   949→   949→        prefix = pattern[:-2]
   950→   950→        return cap_name == prefix or cap_name.startswith(prefix + ".")
   951→   951→    # Match jerárquico: pattern="code" matches cap_name="code.generate".
   952→   952→    if cap_name.startswith(pattern + "."):
   953→   953→        return True
   954→   954→    return False
