     1→     1→"""WorkflowEngine — motor de ejecución de workflows (Sprint 2.2).
     2→     2→
     3→     3→Responsabilidades:
     4→     4→
     5→     5→- Ejecuta un `Workflow` respetando el DAG de dependencias.
     6→     6→- Asigna steps a agentes vía `Scheduler` + `LoadBalancer`.
     7→     7→- Ejecuta steps en paralelo (hasta `max_parallel_steps`).
     8→     8→- Maneja **failover**: retry, reasignación, skip (opcional), cancel.
     9→     9→- Emite eventos al `EventBus` en cada transición.
    10→    10→- Integra el `ContextManager` para compartir contexto entre steps.
    11→    11→- Integra el `AgentLogger` de Sprint 2.1 para telemetría.
    12→    12→- Calcula métricas (duration, parallelism, queue_time, retry_count, success_rate).
    13→    13→
    14→    14→El engine es asyncio-safe y puede ejecutar varios workflows concurrentes.
    15→    15→"""

   115→   115→            "parallelism_max": self.parallelism_max,
   116→   116→            "retry_count": self.retry_count,
   117→   117→            "reassign_count": self.reassign_count,
   118→   118→            "result": self.result,
   119→   119→            "error": self.error,
   120→   120→        }

   127→   127→
   128→   128→class WorkflowEngine:
   129→   129→    """Motor de ejecución de workflows.
   130→   130→
   131→   131→    Args:
   132→   132→        agents:        Lista de agentes disponibles (se actualizará dinámicamente).
   133→   133→        config:        Configuración multi-agent.
   134→   134→        event_bus:     EventBus para telemetría.
   135→   135→        scheduler:     Scheduler (si None, se crea uno).
   136→   136→        load_balancer: LoadBalancer (si None, se crea uno).
   137→   137→        failover:      Política de failover.
   138→   138→        logger:        AgentLogger de Sprint 2.1.
   139→   139→    """
   140→   140→
   141→   141→    def __init__(
   142→   142→        self,
   143→   143→        *,
   144→   144→        agents: list[BaseAgent] | None = None,
   145→   145→        config: MultiAgentConfig | None = None,
   146→   146→        event_bus: EventBus | None = None,
   147→   147→        scheduler: Scheduler | None = None,
   148→   148→        load_balancer: LoadBalancer | None = None,
   149→   149→        failover: FailoverPolicy | None = None,
   150→   150→        agent_logger: AgentLogger | None = None,
   151→   151→        scheduler_config: SchedulerConfig | None = None,
   152→   152→    ) -> None:
   153→   153→        self._config = config or MultiAgentConfig()
   154→   154→        self._bus = event_bus or EventBus()
   155→   155→        self._lb = load_balancer or LoadBalancer()
   156→   156→        self._scheduler = scheduler or Scheduler(
   157→   157→            load_balancer=self._lb,
   158→   158→            event_bus=self._bus,
   159→   159→            config=scheduler_config or SchedulerConfig(),
   160→   160→        )
   161→   161→        self._failover = failover or FailoverPolicy()
   162→   162→        self._logger = agent_logger or AgentLogger(
   163→   163→            history_size=self._config.log_history_size,
   164→   164→        )
   165→   165→        self._agents: dict[str, BaseAgent] = {}
   166→   166→        if agents:
   167→   167→            for a in agents:
   168→   168→                self._agents[a.name] = a
   169→   169→
   170→   170→        # Contextos por workflow.
   171→   171→        self._contexts: dict[str, ContextManager] = {}
   172→   172→        # Workflows en ejecución.
   173→   173→        self._running: dict[str, Workflow] = {}
   174→   174→

   212→   212→    # ------------------------------------------------------------------
   213→   213→    # API principal
   214→   214→    # ------------------------------------------------------------------
   215→   215→
   216→   216→    async def submit(self, workflow: Workflow) -> Workflow:
   217→   217→        """Registra un workflow para ejecución. Emite WORKFLOW_CREATED."""
   218→   218→        # Validar DAG.
   219→   219→        validation = workflow.validate()
   220→   220→        if not validation.ok:
   221→   221→            raise ValueError(
   222→   222→                f"Workflow '{workflow.name}' has invalid DAG: {validation.to_dict()}"
   223→   223→            )
   224→   224→
   225→   225→        self._running[workflow.id] = workflow
   226→   226→        self._contexts[workflow.id] = ContextManager(
   227→   227→            workflow.id, event_bus=self._bus
   228→   228→        )
   229→   229→        await self._emit(
   230→   230→            WorkflowEvent.WORKFLOW_CREATED,
   231→   231→            source="engine",
   232→   232→            payload={
   233→   233→                "workflow_id": workflow.id,
   234→   234→                "name": workflow.name,
   235→   235→                "step_count": workflow.step_count,
   236→   236→                "priority": workflow.priority.value,
   237→   237→            },
   238→   238→            correlation_id=workflow.id,
   239→   239→        )
   240→   240→        return workflow
   241→   241→
   242→   242→    async def run(
   243→   243→        self,
   244→   244→        workflow: Workflow,
   245→   245→        *,
   246→   246→        max_iterations: int = 1000,
   247→   247→    ) -> WorkflowExecutionResult:
   248→   248→        """Ejecuta un workflow completo hasta terminar (o fallar).
   249→   249→
   250→   250→        Bloquea hasta que el workflow reaches terminal state.
   251→   251→        """
   252→   252→        if workflow.id not in self._running:
   253→   253→            await self.submit(workflow)
   254→   254→
   255→   255→        start = time.monotonic()
   256→   256→        result = WorkflowExecutionResult(
   257→   257→            workflow_id=workflow.id,
   258→   258→            steps_total=workflow.step_count,
   259→   259→        )
   260→   260→
   261→   261→        # Estado de asignaciones fallidas para reasignación.
   262→   262→        failed_assignments: dict[str, set[str]] = {}
   263→   263→
   264→   264→        # Control de paralelismo.
   265→   265→        active_steps: dict[str, asyncio.Task] = {}  # step_id -> task
   266→   266→        max_parallelism = 0
   267→   267→
   268→   268→        # Mapeo step_id -> agent_name (para liberar al agente al terminar).
   269→   269→        step_agent_map: dict[str, str] = {}
   270→   270→
   271→   271→        for _ in range(max_iterations):
   272→   272→            # 1. Comprobar si el workflow sigue vivo.
   273→   273→            if workflow.is_terminal:
   274→   274→                break
   275→   275→
   276→   276→            # 2. Planificar nuevos steps.
   277→   277→            decisions = await self._scheduler.schedule(
   278→   278→                workflow,
   279→   279→                list(self._agents.values()),
   280→   280→                exclude=failed_assignments,
   281→   281→            )
   282→   282→
   283→   283→            # 3. Lanzar steps en paralelo.
   284→   284→            for decision in decisions:
   285→   285→                step = workflow.steps.get(decision.step_id)
   286→   286→                if step is None:
   287→   287→                    continue
   288→   288→                agent_name = decision.agent_name
   289→   289→                agent = self._agents.get(agent_name)
   290→   290→                if agent is None:
   291→   291→                    continue
   292→   292→                await self._scheduler.mark_step_started(
   293→   293→                    step, agent_name, workflow
   294→   294→                )
   295→   295→                step_agent_map[step.id] = agent_name
   296→   296→                task = asyncio.create_task(
   297→   297→                    self._execute_step(workflow, step, agent)
   298→   298→                )
   299→   299→                active_steps[step.id] = task
   300→   300→
   301→   301→            max_parallelism = max(max_parallelism, len(active_steps))
   302→   302→
   303→   303→            # 4. Si no hay steps activos ni decisiones, comprobar estado.
   304→   304→            if not active_steps and not decisions:
   305→   305→                # ¿Quedan steps pendientes? Si sí y no hay agentes → fallo.
   306→   306→                pending = [
   307→   307→                    s for s in workflow.steps.values()
   308→   308→                    if s.status in (StepStatus.PENDING, StepStatus.READY)
   309→   309→                ]
   310→   310→                if pending:
   311→   311→                    # Esperar a que un agente quede libre.
   312→   312→                    await asyncio.sleep(0.01)
   313→   313→                    continue
   314→   314→                else:
   315→   315→                    # No hay nada más que hacer.
   316→   316→                    break
   317→   317→
   318→   318→            # 5. Esperar a que termine al menos un step.
   319→   319→            if active_steps:
   320→   320→                done, pending_tasks = await asyncio.wait(
   321→   321→                    active_steps.values(),
   322→   322→                    return_when=asyncio.FIRST_COMPLETED,
   323→   323→                )
   324→   324→                # Procesar completados.
   325→   325→                completed_step_ids: list[str] = []
   326→   326→                for step_id, task in list(active_steps.items()):
   327→   327→                    if task in done:
   328→   328→                        completed_step_ids.append(step_id)
   329→   329→                        try:
   330→   330→                            step_result = await task
   331→   331→                        except Exception as exc:  # noqa: BLE001
   332→   332→                            step_result = {
   333→   333→                                "success": False,
   334→   334→                                "error": str(exc),
   335→   335→                                "duration_ms": 0.0,
   336→   336→                            }
   337→   337→                        agent_name = step_agent_map.pop(step_id, "")
   338→   338→                        await self._handle_step_result(
   339→   339→                            workflow=workflow,
   340→   340→                            step=workflow.steps[step_id],
   341→   341→                            agent_name=agent_name,
   342→   342→                            step_result=step_result,
   343→   343→                            failed_assignments=failed_assignments,
   344→   344→                            result=result,
   345→   345→                        )
   346→   346→                        del active_steps[step_id]
   347→   347→                continue
   348→   348→
   349→   349→        # 6. Workflow terminó (o se agotaron iteraciones).
   350→   350→        result.duration_ms = (time.monotonic() - start) * 1000.0
   351→   351→        result.parallelism_max = max_parallelism
   352→   352→        result.steps_completed = workflow.completed_count
   353→   353→        result.steps_failed = workflow.failed_count
   354→   354→        result.steps_skipped = sum(
   355→   355→            1 for s in workflow.steps.values() if s.status == StepStatus.SKIPPED
   356→   356→        )
   357→   357→        result.status = workflow.status
   358→   358→        result.result = workflow.result
   359→   359→        if workflow.status == WorkflowStatus.FAILED:
   360→   360→            # Buscar el error del primer step REQUIRED fallido.
   361→   361→            for s in workflow.steps.values():
   362→   362→                if s.status == StepStatus.FAILED and s.criticality == StepCriticality.REQUIRED:
   363→   363→                    result.error = s.error
   364→   364→                    break
   365→   365→
   366→   366→        # Emitir evento final.
   367→   367→        final_event = (
   368→   368→            WorkflowEvent.WORKFLOW_COMPLETED
   369→   369→            if workflow.status == WorkflowStatus.COMPLETED
   370→   370→            else WorkflowEvent.WORKFLOW_FAILED
   371→   371→            if workflow.status == WorkflowStatus.FAILED
   372→   372→            else WorkflowEvent.WORKFLOW_CANCELLED
   373→   373→        )
   374→   374→        await self._emit(
   375→   375→            final_event,
   376→   376→            source="engine",
   377→   377→            payload=result.to_dict(),
   378→   378→            correlation_id=workflow.id,
   379→   379→        )
   380→   380→
   381→   381→        return result
   382→   382→
   383→   383→    async def cancel(self, workflow_id: str, reason: str = "") -> bool:
   384→   384→        """Cancela un workflow en ejecución."""
   385→   385→        workflow = self._running.get(workflow_id)
   386→   386→        if workflow is None:
   387→   387→            return False
   388→   388→        if workflow.is_terminal:
   389→   389→            return False
   390→   390→        workflow.transition_to(WorkflowStatus.CANCELLED)
   391→   391→        # Marcar steps no terminales como CANCELLED.
   392→   392→        for step in workflow.steps.values():
   393→   393→            if not step.is_terminal:
   394→   394→                try:
   395→   395→                    step.transition_to(StepStatus.CANCELLED)
   396→   396→                except ValueError:
   397→   397→                    pass
   398→   398→        await self._emit(
   399→   399→            WorkflowEvent.WORKFLOW_CANCELLED,
   400→   400→            source="engine",
   401→   401→            payload={"workflow_id": workflow_id, "reason": reason},
   402→   402→            correlation_id=workflow_id,
   403→   403→        )
   404→   404→        return True
   405→   405→
   406→   406→    async def pause(self, workflow_id: str) -> bool:
   407→   407→        """Pausa un workflow (no interrumpe steps en curso)."""
   408→   408→        workflow = self._running.get(workflow_id)
   409→   409→        if workflow is None or workflow.status != WorkflowStatus.RUNNING:
   410→   410→            return False
   411→   411→        workflow.transition_to(WorkflowStatus.PAUSED)
   412→   412→        await self._emit(
   413→   413→            WorkflowEvent.WORKFLOW_PAUSED,
   414→   414→            source="engine",
   415→   415→            payload={"workflow_id": workflow_id},
   416→   416→            correlation_id=workflow_id,
   417→   417→        )
   418→   418→        return True
   419→   419→
   420→   420→    async def resume(self, workflow_id: str) -> bool:
   421→   421→        """Reanuda un workflow pausado."""
   422→   422→        workflow = self._running.get(workflow_id)
   423→   423→        if workflow is None or workflow.status != WorkflowStatus.PAUSED:
   424→   424→            return False
   425→   425→        workflow.transition_to(WorkflowStatus.RUNNING)
   426→   426→        await self._emit(
   427→   427→            WorkflowEvent.WORKFLOW_RESUMED,
   428→   428→            source="engine",
   429→   429→            payload={"workflow_id": workflow_id},
   430→   430→            correlation_id=workflow_id,
   431→   431→        )
   432→   432→        return True
   433→   433→
   434→   434→    # ------------------------------------------------------------------
   435→   435→    # Ejecución de un step
   436→   436→    # ------------------------------------------------------------------
   437→   437→
   438→   438→    async def _execute_step(
   439→   439→        self,
   440→   440→        workflow: Workflow,
   441→   441→        step: Step,
   442→   442→        agent: BaseAgent,
   443→   443→    ) -> dict[str, Any]:
   444→   444→        """Ejecuta un step en un agente y devuelve el resultado."""
   445→   445→        task_dict = {
   446→   446→            "id": step.id,
   447→   447→            "name": step.name,
   448→   448→            "description": step.action,
   449→   449→            "payload": {**step.input, "action": step.action},
   450→   450→            "priority": workflow.priority.value,
   451→   451→            "workflow_id": workflow.id,
   452→   452→            "context": workflow.context,
   453→   453→        }
   454→   454→        # Log start.
   455→   455→        await self._logger.log_start(
   456→   456→            agent.name, step.id, f"workflow step: {step.name}"
   457→   457→        )
   458→   458→        start = time.monotonic()
   459→   459→        result = await agent.execute(task_dict)
   460→   460→        duration_ms = (time.monotonic() - start) * 1000.0
   461→   461→        success = result.get("status") == "ok"
   462→   462→        # Log end.
   463→   463→        if success:
   464→   464→            inner = result.get("result", {})
   465→   465→            await self._logger.log_end(
   466→   466→                agent.name, step.id,
   467→   467→                duration_ms=duration_ms,
   468→   468→                tokens_prompt=int(inner.get("tokens_prompt", 0)) if isinstance(inner, dict) else 0,
   469→   469→                tokens_completion=int(inner.get("tokens_completion", 0)) if isinstance(inner, dict) else 0,
   470→   470→                tools_used=inner.get("tools_used", []) if isinstance(inner, dict) else [],
   471→   471→            )
   472→   472→            # Registrar herramientas usadas.
   473→   473→            if isinstance(inner, dict):
   474→   474→                for tool in inner.get("tools_used", []):
   475→   475→                    await self._emit(
   476→   476→                        WorkflowEvent.TOOL_EXECUTED,
   477→   477→                        source=agent.name,
   478→   478→                        payload={
   479→   479→                            "workflow_id": workflow.id,
   480→   480→                            "step_id": step.id,
   481→   481→                            "tool": tool,
   482→   482→                        },
   483→   483→                        correlation_id=workflow.id,
   484→   484→                    )
   485→   485→            # Actualizar step.
   486→   486→            step.output = inner
   487→   487→            step.record_execution(
   488→   488→                duration_ms=duration_ms,
   489→   489→                tokens_prompt=int(inner.get("tokens_prompt", 0)) if isinstance(inner, dict) else 0,
   490→   490→                tokens_completion=int(inner.get("tokens_completion", 0)) if isinstance(inner, dict) else 0,
   491→   491→                tools_used=inner.get("tools_used", []) if isinstance(inner, dict) else None,
   492→   492→                files_modified=inner.get("files_modified", []) if isinstance(inner, dict) else None,
   493→   493→            )
   494→   494→            # Actualizar contexto.
   495→   495→            ctx = self._contexts.get(workflow.id)
   496→   496→            if ctx is not None and isinstance(inner, dict):
   497→   497→                if "files_modified" in inner:
   498→   498→                    for path in inner["files_modified"]:
   499→   499→                        await ctx.register_file(path, None, modified_by=agent.name)
   500→   500→        else:
   501→   501→            await self._logger.log_error(
   502→   502→                agent.name, step.id,
   503→   503→                result.get("error", "unknown"),
   504→   504→            )
   505→   505→        return {
   506→   506→            "success": success,
   507→   507→            "error": result.get("error"),
   508→   508→            "duration_ms": duration_ms,
   509→   509→            "result": result.get("result"),
   510→   510→        }
   511→   511→
   512→   512→    # ------------------------------------------------------------------
   513→   513→    # Manejo del resultado de un step (failover)
   514→   514→    # ------------------------------------------------------------------
   515→   515→
   516→   516→    async def _handle_step_result(
   517→   517→        self,
   518→   518→        *,
   519→   519→        workflow: Workflow,
   520→   520→        step: Step,
   521→   521→        agent_name: str,
   522→   522→        step_result: dict[str, Any],
   523→   523→        failed_assignments: dict[str, set[str]],
   524→   524→        result: WorkflowExecutionResult,
   525→   525→    ) -> None:
   526→   526→        """Procesa el resultado de un step y decide el failover."""
   527→   527→        success = step_result.get("success", False)
   528→   528→        duration_ms = step_result.get("duration_ms", 0.0)
   529→   529→        error = step_result.get("error")
   530→   530→
   531→   531→        await self._scheduler.mark_step_finished(
   532→   532→            step, agent_name, workflow,
   533→   533→            success=success,
   534→   534→            duration_ms=duration_ms,
   535→   535→            error=error if not success else None,
   536→   536→        )
   537→   537→
   538→   538→        if success:
   539→   539→            # ¿Workflow completo?
   540→   540→            if workflow.progress >= 1.0:
   541→   541→                workflow.transition_to(WorkflowStatus.COMPLETED)
   542→   542→                workflow.result = step_result.get("result")
   543→   543→            return
   544→   544→
   545→   545→        # Failover.
   546→   546→        action = self._decide_failover(step, failed_assignments)
   547→   547→        if action == FailoverAction.RETRY:
   548→   548→            step.retry += 1
   549→   549→            step.error = None
   550→   550→            step.transition_to(StepStatus.READY)
   551→   551→            # Resetear el agente a IDLE para que pueda aceptar el retry
   552→   552→            # (tras un fallo, BaseAgent.execute() lo deja en ERROR).
   553→   553→            agent = self._agents.get(agent_name)
   554→   554→            if agent is not None and agent.status == AgentStatus.ERROR:
   555→   555→                agent._set_status(AgentStatus.IDLE)  # noqa: SLF001
   556→   556→            result.retry_count += 1
   557→   557→            await self._emit(
   558→   558→                WorkflowEvent.STEP_RETRY,
   559→   559→                source="engine",
   560→   560→                payload={
   561→   561→                    "workflow_id": workflow.id,
   562→   562→                    "step_id": step.id,
   563→   563→                    "attempt": step.retry,
   564→   564→                },
   565→   565→                correlation_id=workflow.id,
   566→   566→            )
   567→   567→        elif action == FailoverAction.REASSIGN:
   568→   568→            # Marcar el agente como fallido para este step.
   569→   569→            failed_assignments.setdefault(step.id, set()).add(agent_name)
   570→   570→            step.agent_reassigned.append(agent_name)
   571→   571→            step.error = None
   572→   572→            step.transition_to(StepStatus.READY)
   573→   573→            # Resetear el agente (queda en ERROR tras el fallo).
   574→   574→            agent = self._agents.get(agent_name)
   575→   575→            if agent is not None and agent.status == AgentStatus.ERROR:
   576→   576→                agent._set_status(AgentStatus.IDLE)  # noqa: SLF001
   577→   577→            result.reassign_count += 1
   578→   578→            await self._emit(
   579→   579→                WorkflowEvent.AGENT_REASSIGNED,
   580→   580→                source="engine",
   581→   581→                payload={
   582→   582→                    "workflow_id": workflow.id,
   583→   583→                    "step_id": step.id,
   584→   584→                    "from_agent": agent_name,
   585→   585→                    "excluded": list(failed_assignments[step.id]),
   586→   586→                },
   587→   587→                correlation_id=workflow.id,
   588→   588→            )
   589→   589→        elif action == FailoverAction.SKIP:
   590→   590→            step.transition_to(StepStatus.SKIPPED)
   591→   591→            await self._emit(
   592→   592→                WorkflowEvent.STEP_SKIPPED,
   593→   593→                source="engine",
   594→   594→                payload={
   595→   595→                    "workflow_id": workflow.id,
   596→   596→                    "step_id": step.id,
   597→   597→                    "reason": "optional step failed",
   598→   598→                },
   599→   599→                correlation_id=workflow.id,
   600→   600→            )
   601→   601→            # Propagar SKIP a dependientes (no pueden ejecutarse sin su dep).
   602→   602→            self._propagate_skip(workflow, step.id)
   603→   603→            # ¿Workflow completo tras skip?
   604→   604→            if self._all_steps_terminal(workflow):
   605→   605→                required_failed = any(
   606→   606→                    s.status == StepStatus.FAILED and s.criticality == StepCriticality.REQUIRED
   607→   607→                    for s in workflow.steps.values()
   608→   608→                )
   609→   609→                if not required_failed:
   610→   610→                    workflow.transition_to(WorkflowStatus.COMPLETED)
   611→   611→                    workflow.result = {"skipped_steps": [
   612→   612→                        s.id for s in workflow.steps.values() if s.status == StepStatus.SKIPPED
   613→   613→                    ]}
   614→   614→        else:  # NONE or CANCEL
   615→   615→            # Marcar como fallo definitivo.
   616→   616→            # Si el step es REQUIRED y cancel_on_required_fail, cancelar workflow.
   617→   617→            if (
   618→   618→                step.criticality == StepCriticality.REQUIRED
   619→   619→                and self._failover.cancel_on_required_fail
   620→   620→            ):
   621→   621→                workflow.transition_to(WorkflowStatus.FAILED)
   622→   622→                # Propagar SKIP a dependientes (no pueden ejecutarse).
   623→   623→                self._propagate_skip(workflow, step.id)
   624→   624→            else:
   625→   625→                # Step OPTIONAL fallido definitivo: dejarlo FAILED pero
   626→   626→                # permitir que el workflow continúe.
   627→   627→                # Propagar SKIP a dependientes.
   628→   628→                self._propagate_skip(workflow, step.id)
   629→   629→                if self._all_steps_terminal(workflow):
   630→   630→                    required_failed = any(
   631→   631→                        s.status == StepStatus.FAILED and s.criticality == StepCriticality.REQUIRED
   632→   632→                        for s in workflow.steps.values()
   633→   633→                    )
   634→   634→                    if not required_failed and workflow.completed_count > 0:
   635→   635→                        workflow.transition_to(WorkflowStatus.COMPLETED)
   636→   636→                    elif required_failed:
   637→   637→                        workflow.transition_to(WorkflowStatus.FAILED)
   638→   638→
   639→   639→    def _decide_failover(
   640→   640→        self,
   641→   641→        step: Step,
   642→   642→        failed_assignments: dict[str, set[str]],
   643→   643→    ) -> FailoverAction:
   644→   644→        """Decide la acción de failover para un step fallido."""
   645→   645→        # 1. Retry si quedan intentos.
   646→   646→        if step.retry < self._failover.max_retries:
   647→   647→            return FailoverAction.RETRY
   648→   648→        # 2. Reasignar si está habilitado y no hemos excedido el máximo.
   649→   649→        if (
   650→   650→            self._failover.reassign_on_fail
   651→   651→            and len(step.agent_reassigned) < self._failover.max_reassigns
   652→   652→        ):
   653→   653→            return FailoverAction.REASSIGN
   654→   654→        # 3. Skip si es opcional.
   655→   655→        if (
   656→   656→            step.criticality == StepCriticality.OPTIONAL
   657→   657→            and self._failover.skip_optional_on_fail
   658→   658→        ):
   659→   659→            return FailoverAction.SKIP
   660→   660→        # 4. Cancelar workflow si es required y cancel_on_required_fail.
   661→   661→        if (
   662→   662→            step.criticality == StepCriticality.REQUIRED
   663→   663→            and self._failover.cancel_on_required_fail
   664→   664→        ):
   665→   665→            return FailoverAction.CANCEL
   666→   666→        return FailoverAction.NONE
   667→   667→
   668→   668→    def _all_steps_terminal(self, workflow: Workflow) -> bool:
   669→   669→        """Devuelve True si todos los steps están en estado terminal."""
   670→   670→        return all(s.is_terminal for s in workflow.steps.values())
   671→   671→
   672→   672→    def _propagate_skip(self, workflow: Workflow, skipped_step_id: str) -> None:
   673→   673→        """Propaga SKIP a todos los dependientes del step saltado.
   674→   674→
   675→   675→        Esto evita que el engine intente ejecutar steps cuyas dependencias
   676→   676→        no se completaron. Los dependientes se marcan como SKIPPED en cadena.
   677→   677→        """
   678→   678→        dependents = workflow.dag.dependents_of(skipped_step_id)
   679→   679→        for dep_id in dependents:
   680→   680→            dep_step = workflow.steps.get(dep_id)
   681→   681→            if dep_step is None or dep_step.is_terminal:
   682→   682→                continue
   683→   683→            try:
   684→   684→                dep_step.transition_to(StepStatus.SKIPPED)
   685→   685→                # Recursión para propagar en cadena.
   686→   686→                self._propagate_skip(workflow, dep_id)
   687→   687→            except ValueError:
   688→   688→                # El step ya está en un estado que no permite SKIP.
   689→   689→                pass
   690→   690→

   696→   696→        """Métricas agregadas del engine."""

   702→   702→            "running_workflows": len(self._running),
   703→   703→            "registered_agents": len(self._agents),
   704→   704→        }

   710→   710→            return None
