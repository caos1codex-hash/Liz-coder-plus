     1→"""AgentRegistry — registro único de agentes (Sprint 2.2/2).
     2→
     3→Responsabilidades:
     4→
     5→- `register(agent)` / `unregister(id_or_name)` / `replace(old, new)`
     6→- `discover(...)` — auto-discovery de agentes
     7→- `get(id_or_name)` / `get_all()` / `exists(id_or_name)`
     8→- `health()` — health check de todos los agentes
     9→- `metrics()` — métricas del registry
    10→
    11→Cada agente registrado se envuelve en un `AgentRecord` que añade:
    12→
    13→- id (UUID), name, version, description
    14→- capabilities (provided/required/preferred)
    15→- priority, status, metadata
    16→- created_at, last_seen, last_heartbeat
    17→- health (HealthState), errors, uptime, usage
    18→
    19→El registry es **asyncio-safe** y emite eventos al `EventBus`:
    20→
    21→- `agent.registered`
    22→- `agent.removed`
    23→- `agent.heartbeat.received`
    24→- `agent.health.changed`
    25→
    26→El registry **no ejecuta agentes**; sólo los trackea. La selección
    27→se hace vía `CapabilityResolver`.
    28→"""
    29→
    30→from __future__ import annotations
    31→
    32→import asyncio
    33→import logging
    34→import time
    35→import uuid
    36→from dataclasses import dataclass, field
    37→from datetime import datetime, timezone
    38→from typing import TYPE_CHECKING, Any
    39→
    40→from ..base import BaseAgent
    41→from ..enums import AgentStatus, HealthState
    42→from .capability import Capability, CapabilityRegistry
    43→
    44→if TYPE_CHECKING:
    45→    from .manifest import AgentManifest
    46→
    47→logger = logging.getLogger(__name__)
    48→
    49→
    50→def _now_iso() -> str:
    51→    return datetime.now(timezone.utc).isoformat()
    52→
    53→
    54→def _now_epoch() -> float:
    55→    return time.time()
    56→
    57→
    58→# ----------------------------------------------------------------------
    59→# AgentRecord
    60→# ----------------------------------------------------------------------
    61→
    62→
    63→@dataclass
    64→class AgentRecord:
    65→    """Wrapper sobre un BaseAgent con metadatos de registro.
    66→
    67→    Attributes:
    68→        agent:           El BaseAgent registrado.
    69→        id:              UUID único del registro.
    70→        name:            Nombre del agente (debe ser único en el registry).
    71→        version:         Versión del agente.
    72→        description:     Descripción humana.
    73→        provided_capabilities: Capabilities que el agente provee.
    74→        required_capabilities: Capabilities que el agente necesita de otros.
    75→        preferred_capabilities: Capabilities preferidas (para routing suave).
    76→        priority:        Prioridad del agente (mayor = más prioritario).
    77→        status:          AgentStatus actual (snapshot).
    78→        metadata:        Metadatos libres.
    79→        created_at:      ISO timestamp de registro.
    80→        last_seen:       ISO timestamp de última actividad.
    81→        last_heartbeat:  ISO timestamp del último heartbeat.
    82→        health_state:    HealthState (alive/busy/idle/error/dead).
    83→        errors:          Lista de errores recientes (máx 20).
    84→        uptime_s:        Segundos desde created_at.
    85→        usage_count:     Número de veces que se seleccionó.
    86→        success_count:   Número de selecciones exitosas.
    87→        failure_count:   Número de selecciones fallidas.
    88→        total_duration_ms: Tiempo total ocupado (ms).
    89→        avg_latency_ms:  Latencia media de las últimas N ejecuciones.
    90→    """
    91→
    92→    agent: BaseAgent
    93→    id: str = field(default_factory=lambda: str(uuid.uuid4()))
    94→    name: str = ""
    95→    version: str = "0.1.0"
    96→    description: str = ""
    97→    provided_capabilities: list[str] = field(default_factory=list)
    98→    required_capabilities: list[str] = field(default_factory=list)
    99→    preferred_capabilities: list[str] = field(default_factory=list)
   100→    priority: int = 0
   101→    status: AgentStatus = AgentStatus.CREATED
   102→    metadata: dict[str, Any] = field(default_factory=dict)
   103→    created_at: str = field(default_factory=_now_iso)
   104→    last_seen: str = field(default_factory=_now_iso)
   105→    last_heartbeat: str = field(default_factory=_now_iso)
   106→    health_state: HealthState = HealthState.ALIVE
   107→    errors: list[str] = field(default_factory=list)
   108→    usage_count: int = 0
   109→    success_count: int = 0
   110→    failure_count: int = 0
   111→    total_duration_ms: float = 0.0
   112→    _latencies: list[float] = field(default_factory=list)
   113→
   114→    MAX_ERRORS = 20
   115→    MAX_LATENCIES = 100
   116→
   117→    def __post_init__(self) -> None:
   118→        if not self.name:
   119→            self.name = self.agent.name
   120→        if not self.description:
   121→            self.description = self.agent.description
   122→        if not self.version:
   123→            self.version = getattr(self.agent, "version", "0.1.0")
   124→
   125→    # ------------------------------------------------------------------
   126→    # Health / heartbeat
   127→    # ------------------------------------------------------------------
   128→
   129→    def heartbeat(self) -> None:
   130→        """Actualiza last_heartbeat y last_seen."""
   131→        self.last_heartbeat = _now_iso()
   132→        self.last_seen = _now_iso()
   133→
   134→    def update_status(self) -> None:
   135→        """Snapshot del estado del agente subyacente."""
   136→        self.status = self.agent.status
   137→        report = self.agent.health() if hasattr(self.agent, "health") else None
   138→        if report is not None:
   139→            self.health_state = report.state
   140→
   141→    def record_usage(
   142→        self,
   143→        *,
   144→        success: bool,
   145→        duration_ms: float = 0.0,
   146→    ) -> None:
   147→        """Registra una selección del agente."""
   148→        self.usage_count += 1
   149→        self.total_duration_ms += duration_ms
   150→        self._latencies.append(duration_ms)
   151→        if len(self._latencies) > self.MAX_LATENCIES:
   152→            self._latencies = self._latencies[-self.MAX_LATENCIES:]
   153→        if success:
   154→            self.success_count += 1
   155→        else:
   156→            self.failure_count += 1
   157→            # Sincronizar errores del agente si los tiene.
   158→            last_err = getattr(self.agent, "last_error", None)
   159→            if last_err:
   160→                self.add_error(last_err)
   161→        self.last_seen = _now_iso()
   162→
   163→    def add_error(self, error: str) -> None:
   164→        """Añade un error al historial (con cap)."""
   165→        self.errors.append(error)
   166→        if len(self.errors) > self.MAX_ERRORS:
   167→            self.errors = self.errors[-self.MAX_ERRORS:]
   168→
   169→    @property
   170→    def avg_latency_ms(self) -> float:
   171→        if not self._latencies:
   172→            return 0.0
   173→        return sum(self._latencies) / len(self._latencies)
   174→
   175→    @property
   176→    def success_rate(self) -> float:
   177→        total = self.success_count + self.failure_count
   178→        if total == 0:
   179→            return 1.0
   180→        return self.success_count / total
   181→
   182→    @property
   183→    def uptime_s(self) -> float:
   184→        """Segundos desde created_at (aproximado, basado en epoch)."""
   185→        try:
   186→            created = datetime.fromisoformat(self.created_at).timestamp()
   187→            return _now_epoch() - created
   188→        except (ValueError, OSError):
   189→            return 0.0
   190→
   191→    @property
   192→    def is_healthy(self) -> bool:
   193→        """True si health_state es ALIVE o IDLE."""
   194→        return self.health_state in (HealthState.ALIVE, HealthState.IDLE)
   195→
   196→    @property
   197→    def is_available(self) -> bool:
   198→        """True si está disponible para selección (no DEAD/ERROR/STOPPED)."""
   199→        return (
   200→            self.status in (AgentStatus.IDLE, AgentStatus.CREATED)
   201→            and self.health_state in (HealthState.ALIVE, HealthState.IDLE)
   202→        )
   203→
   204→    # ------------------------------------------------------------------
   205→    # Capabilities
   206→    # ------------------------------------------------------------------
   207→
   208→    def provides(self, capability: str) -> bool:
   209→        """True si el agente provee la capability (con wildcard match)."""
   210→        for cap in self.provided_capabilities:
   211→            # Match exacto.
   212→            if cap == capability:
   213→                return True
   214→            # Match wildcard: cap="code.*" matches capability="code.generate".
   215→            if cap.endswith(".*"):
   216→                prefix = cap[:-2]
   217→                if capability == prefix or capability.startswith(prefix + "."):
   218→                    return True
   219→            # Match jerárquico: cap="code" matches capability="code.generate"
   220→            # (el agente que provee "code" puede manejar sub-capabilities).
   221→            if capability.startswith(cap + "."):
   222→                return True
   223→        return False
   224→
   225→    def requires_satisfied(self, available_caps: set[str]) -> bool:
   226→        """True si todas las capabilities requeridas están disponibles."""
   227→        for req in self.required_capabilities:
   228→            if not any(
   229→                req == c or req.startswith(c + ".") or
   230→                (c.endswith(".*") and (req == c[:-2] or req.startswith(c[:-2] + ".")))
   231→                for c in available_caps
   232→            ):
   233→                return False
   234→        return True
   235→
   236→    # ------------------------------------------------------------------
   237→    # Serialización
   238→    # ------------------------------------------------------------------
   239→
   240→    def to_dict(self) -> dict[str, Any]:
   241→        return {
   242→            "id": self.id,
   243→            "name": self.name,
   244→            "version": self.version,
   245→            "description": self.description,
   246→            "provided_capabilities": list(self.provided_capabilities),
   247→            "required_capabilities": list(self.required_capabilities),
   248→            "preferred_capabilities": list(self.preferred_capabilities),
   249→            "priority": self.priority,
   250→            "status": self.status.value,
   251→            "metadata": dict(self.metadata),
   252→            "created_at": self.created_at,
   253→            "last_seen": self.last_seen,
   254→            "last_heartbeat": self.last_heartbeat,
   255→            "health_state": self.health_state.value,
   256→            "errors": list(self.errors),
   257→            "uptime_s": round(self.uptime_s, 2),
   258→            "usage_count": self.usage_count,
   259→            "success_count": self.success_count,
   260→            "failure_count": self.failure_count,
   261→            "success_rate": round(self.success_rate, 4),
   262→            "total_duration_ms": round(self.total_duration_ms, 2),
   263→            "avg_latency_ms": round(self.avg_latency_ms, 2),
   264→        }

   267→# ----------------------------------------------------------------------
   268→# AgentRegistry
   269→# ----------------------------------------------------------------------
   270→
   271→
   272→class AgentRegistry:
   273→    """Registro único de agentes.
   274→
   275→    Uso::
   276→
   277→        registry = AgentRegistry(capability_registry=cap_reg, event_bus=bus)
   278→        await registry.register(my_agent, provided_capabilities=["code.generate"])
   279→        record = await registry.get_by_name("my_agent")
   280→        candidates = registry.find_by_capability("code.generate")
   281→    """
   282→
   283→    HEARTBEAT_TIMEOUT_S = 60.0  # sin heartbeat en 60s → unhealthy.
   284→
   285→    def __init__(
   286→        self,
   287→        *,
   288→        capability_registry: CapabilityRegistry | None = None,
   289→        event_bus: Any | None = None,
   290→        heartbeat_timeout: float = 60.0,
   291→    ) -> None:
   292→        self._records: dict[str, AgentRecord] = {}  # by id
   293→        self._by_name: dict[str, str] = {}  # name -> record_id
   294→        self._cap_registry = capability_registry or CapabilityRegistry()
   295→        self._bus = event_bus
   296→        self._heartbeat_timeout = heartbeat_timeout
   297→        self._lock = asyncio.Lock()
   298→        self._heartbeat_task: asyncio.Task[None] | None = None
   299→        self._auto_discovery_hooks: list = []
   300→
   301→        # Métricas.
   302→        self._metrics = {
   303→            "registered_total": 0,
   304→            "unregistered_total": 0,
   305→            "replaced_total": 0,
   306→            "discoveries_total": 0,
   307→            "heartbeat_checks": 0,
   308→            "health_changes": 0,
   309→        }
   310→
   311→    @property
   312→    def capability_registry(self) -> CapabilityRegistry:
   313→        return self._cap_registry
   314→
   315→    @property
   316→    def event_bus(self) -> Any | None:
   317→        return self._bus
   318→
   319→    # ------------------------------------------------------------------
   320→    # Registro
   321→    # ------------------------------------------------------------------
   322→
   323→    async def register(
   324→        self,
   325→        agent: BaseAgent,
   326→        *,
   327→        provided_capabilities: list[str] | None = None,
   328→        required_capabilities: list[str] | None = None,
   329→        preferred_capabilities: list[str] | None = None,
   330→        priority: int = 0,
   331→        metadata: dict[str, Any] | None = None,
   332→        version: str = "0.1.0",
   333→        replace_existing: bool = True,
   334→    ) -> AgentRecord:
   335→        """Registra un agente.
   336→
   337→        Args:
   338→            agent:                   BaseAgent a registrar.
   339→            provided_capabilities:   Capabilities que provee.
   340→            required_capabilities:   Capabilities que necesita de otros.
   341→            preferred_capabilities:  Capabilities preferidas.
   342→            priority:                Prioridad (mayor = más prioritario).
   343→            metadata:                Metadatos libres.
   344→            version:                 Versión del agente.
   345→            replace_existing:        Si True y ya existe un agente con el
   346→                                     mismo nombre, lo reemplaza.
   347→
   348→        Returns:
   349→            El AgentRecord creado.
   350→
   351→        Raises:
   352→            ValueError: Si ya existe un agente con el mismo nombre y
   353→                replace_existing=False.
   354→            TypeError: Si el objeto no parece un BaseAgent.
   355→        """
   356→        # Duck-typing: verificar atributos clave en lugar de isinstance,
   357→        # para ser robusto frente a re-imports del módulo durante los
   358→        # tests (que crean clases BaseAgent distintas con el mismo nombre).
   359→        if not (
   360→            hasattr(agent, "name")
   361→            and hasattr(agent, "status")
   362→            and hasattr(agent, "health")
   363→            and hasattr(agent, "execute")
   364→            and hasattr(agent, "start")
   365→            and hasattr(agent, "stop")
   366→        ):
   367→            raise TypeError(
   368→                f"Expected BaseAgent (or compatible), got {type(agent).__name__}"
   369→            )
   370→
   371→        async with self._lock:
   372→            existing_id = self._by_name.get(agent.name)
   373→            if existing_id is not None:
   374→                if not replace_existing:
   375→                    raise ValueError(
   376→                        f"Agent '{agent.name}' already registered "
   377→                        f"(id={existing_id})"
   378→                    )
   379→                # Reemplazar.
   380→                old_record = self._records.pop(existing_id, None)
   381→                self._by_name.pop(agent.name, None)
   382→                self._metrics["replaced_total"] += 1
   383→                if old_record is not None:
   384→                    await self._emit("agent.removed", {
   385→                        "agent_id": old_record.id,
   386→                        "name": old_record.name,
   387→                        "reason": "replaced",
   388→                    })
   389→
   390→            record = AgentRecord(
   391→                agent=agent,
   392→                name=agent.name,
   393→                description=agent.description,
   394→                version=version,
   395→                provided_capabilities=list(provided_capabilities or []),
   396→                required_capabilities=list(required_capabilities or []),
   397→                preferred_capabilities=list(preferred_capabilities or []),
   398→                priority=priority,
   399→                metadata=dict(metadata or {}),
   400→            )
   401→            record.update_status()
   402→            record.heartbeat()
   403→            self._records[record.id] = record
   404→            self._by_name[record.name] = record.id
   405→            self._metrics["registered_total"] += 1
   406→
   407→        await self._emit("agent.registered", {
   408→            "agent_id": record.id,
   409→            "name": record.name,
   410→            "version": record.version,
   411→            "provided_capabilities": record.provided_capabilities,
   412→            "priority": record.priority,
   413→        })
   414→        logger.info(
   415→            "Registered agent '%s' (id=%s, caps=%d, priority=%d)",
   416→            record.name, record.id[:8],
   417→            len(record.provided_capabilities), record.priority,
   418→        )
   419→        return record
   420→
   421→    async def unregister(self, id_or_name: str) -> bool:
   422→        """Elimina un agente por id o nombre."""
   423→        async with self._lock:
   424→            record = self._resolve(id_or_name)
   425→            if record is None:
   426→                return False
   427→            self._records.pop(record.id, None)
   428→            self._by_name.pop(record.name, None)
   429→            self._metrics["unregistered_total"] += 1
   430→
   431→        await self._emit("agent.removed", {
   432→            "agent_id": record.id,
   433→            "name": record.name,
   434→            "reason": "unregistered",
   435→        })
   436→        logger.info("Unregistered agent '%s' (id=%s)", record.name, record.id[:8])
   437→        return True
   438→
   439→    async def replace(
   440→        self,
   441→        old_id_or_name: str,
   442→        new_agent: BaseAgent,
   443→        **kwargs: Any,
   444→    ) -> AgentRecord:
   445→        """Reemplaza un agente existente por uno nuevo.
   446→
   447→        Mantiene el mismo record_id si el nombre coincide; si no,
   448→        crea un nuevo registro.
   449→        """
   450→        async with self._lock:
   451→            old_record = self._resolve(old_id_or_name)
   452→            if old_record is None:
   453→                raise KeyError(f"Agent '{old_id_or_name}' not found")
   454→            # Si el nuevo agente tiene el mismo nombre, reusar el id.
   455→            if new_agent.name == old_record.name:
   456→                kwargs.setdefault("priority", old_record.priority)
   457→                kwargs.setdefault("metadata", old_record.metadata)
   458→                kwargs.setdefault("version", old_record.version)
   459→                # Eliminar el viejo (sin emitir evento de removed).
   460→                self._records.pop(old_record.id, None)
   461→                self._by_name.pop(old_record.name, None)
   462→
   463→        # Registrar el nuevo (replace_existing=True para evitar conflictos).
   464→        return await self.register(new_agent, replace_existing=True, **kwargs)
   465→
   466→    # ------------------------------------------------------------------
   467→    # Auto-discovery
   468→    # ------------------------------------------------------------------
   469→
   470→    def add_discovery_hook(self, hook) -> None:
   471→        """Añade un hook de auto-discovery.
   472→
   473→        El hook es un callable async `(registry) -> list[BaseAgent]` que
   474→        devuelve agentes descubiertos. El registry los registrará.
   475→        """
   476→        self._auto_discovery_hooks.append(hook)
   477→
   478→    async def discover(self) -> list[AgentRecord]:
   479→        """Ejecuta todos los hooks de discovery y registra los agentes.
   480→
   481→        Returns:
   482→            Lista de AgentRecord registrados (puede estar vacía).
   483→        """
   484→        discovered: list[AgentRecord] = []
   485→        for hook in self._auto_discovery_hooks:
   486→            try:
   487→                agents = await hook(self)
   488→            except Exception:  # noqa: BLE001
   489→                logger.exception("Discovery hook failed")
   490→                continue
   491→            for agent in agents:
   492→                if not isinstance(agent, BaseAgent):
   493→                    logger.warning(
   494→                        "Discovery hook returned non-BaseAgent: %s",
   495→                        type(agent).__name__,
   496→                    )
   497→                    continue
   498→                # Auto-detect capabilities from agent's objectives/tools.
   499→                caps = self._infer_capabilities(agent)
   500→                try:
   501→                    record = await self.register(
   502→                        agent,
   503→                        provided_capabilities=caps,
   504→                        replace_existing=True,
   505→                    )
   506→                    discovered.append(record)
   507→                    self._metrics["discoveries_total"] += 1
   508→                except Exception:  # noqa: BLE001
   509→                    logger.exception(
   510→                        "Failed to register discovered agent '%s'",
   511→                        agent.name,
   512→                    )
   513→        if discovered:
   514→            logger.info("Discovered and registered %d agents", len(discovered))
   515→        return discovered
   516→
   517→    def _infer_capabilities(self, agent: BaseAgent) -> list[str]:
   518→        """Infiere capabilities de un agente a partir de sus objetivos y tools.
   519→
   520→        Heurística simple: si el agente tiene `default_objectives` o
   521→        `objectives` que coinciden con categorías de capabilities, las
   522→        deriva. Si tiene `tools`, los mapea a capabilities.
   523→        """
   524→        caps: list[str] = []
   525→        objectives = list(getattr(agent, "objectives", []) or [])
   526→        tools = list(getattr(agent, "tools", []) or [])
   527→
   528→        # Mapping heurístico objetivo → capability.
   529→        obj_to_cap = {
   530→            "plan": "planning",
   531→            "planning": "planning",
   532→            "decompose": "planning",
   533→            "workflow": "workflow",
   534→            "code": "code.generate",
   535→            "refactor": "code.refactor",
   536→            "fix": "code.fix",
   537→            "test": "code.test",
   538→            "implement": "code.generate",
   539→            "review": "code.review",
   540→            "audit": "code.review",
   541→            "security": "code.review",
   542→            "git": "git.commit",
   543→            "commit": "git.commit",
   544→            "branch": "git.branch",
   545→            "merge": "git.merge",
   546→            "push": "git.push",
   547→            "terminal": "terminal.execute",
   548→            "execute": "terminal.execute",
   549→            "shell": "terminal.execute",
   550→            "memory": "memory.store",
   551→            "store": "memory.store",
   552→            "retrieve": "memory.retrieve",
   553→            "research": "research",
   554→            "investigate": "research",
   555→            "summarize": "research.summarize",
   556→            "search": "web.search",
   557→        }
   558→        for obj in objectives:
   559→            obj_lower = obj.lower()
   560→            if obj_lower in obj_to_cap:
   561→                cap = obj_to_cap[obj_lower]
   562→                if cap not in caps:
   563→                    caps.append(cap)
   564→
   565→        # Mapping heurístico tool → capability.
   566→        tool_to_cap = {
   567→            "file_tool": "filesystem.read",
   568→            "terminal_tool": "terminal.execute",
   569→            "system_tool": "terminal.execute",
   570→            "git": "git.commit",
   571→            "memory_store": "memory.store",
   572→            "vector_index": "memory.search",
   573→            "web_search": "web.search",
   574→            "doc_fetch": "documentation.lookup",
   575→            "planner": "planning",
   576→            "dag_builder": "workflow.create",
   577→            "linter": "code.review",
   578→            "static_analyzer": "code.review",
   579→            "refactor": "code.refactor",
   580→            "fix": "code.fix",
   581→            "test_gen": "code.test",
   582→        }
   583→        for tool in tools:
   584→            tool_lower = tool.lower()
   585→            if tool_lower in tool_to_cap:
   586→                cap = tool_to_cap[tool_lower]
   587→                if cap not in caps:
   588→                    caps.append(cap)
   589→
   590→        return caps

   623→    def find_by_capability(
   624→        self,
   625→        capability: str,
   626→        *,
   627→        only_available: bool = True,
   628→    ) -> list[AgentRecord]:
   629→        """Devuelve agentes que proveen una capability.
   630→
   631→        Args:
   632→            capability:      Nombre o patrón de la capability.
   633→            only_available:  Si True, sólo devuelve agentes disponibles
   634→                             (IDLE/CREATED + healthy).
   635→        """
   636→        result: list[AgentRecord] = []
   637→        for record in self._records.values():
   638→            if only_available and not record.is_available:
   639→                continue
   640→            if record.provides(capability):
   641→                result.append(record)
   642→        # Ordenar por prioridad (desc) y luego por usage_count (asc,
   643→        # para balanceo).
   644→        result.sort(key=lambda r: (-r.priority, r.usage_count))
   645→        return result

   647→    def find_by_capability_pattern(
   648→        self,
   649→        pattern: str,
   650→        *,
   651→        only_available: bool = True,
   652→    ) -> list[AgentRecord]:
   653→        """Como find_by_capability pero con soporte de wildcards.
   654→
   655→        Ej: "code.*" devuelve agentes que proveen code.generate,
   656→        code.review, etc.
   657→        """
   658→        if pattern == "*":
   659→            return [
   660→                r for r in self._records.values()
   661→                if not only_available or r.is_available
   662→            ]
   663→        # Expandir el patrón contra las provided_capabilities de cada agente.
   664→        result: list[AgentRecord] = []
   665→        for record in self._records.values():
   666→            if only_available and not record.is_available:
   667→                continue
   668→            for cap in record.provided_capabilities:
   669→                cap_obj = Capability(name=cap)
   670→                if cap_obj.matches(pattern) or _pattern_matches_cap(pattern, cap):
   671→                    result.append(record)
   672→                    break
   673→        result.sort(key=lambda r: (-r.priority, r.usage_count))
   674→        return result

   680→    def heartbeat(self, id_or_name: str) -> bool:
   681→        """Registra un heartbeat para un agente."""

   688→    def update_health(self) -> int:
   689→        """Actualiza el estado de salud de todos los agentes.
   690→
   691→        Returns el número de agentes cuyo health_state cambió.
   692→        """
   693→        changes = 0
   694→        now_epoch = _now_epoch()
   695→        for record in self._records.values():
   696→            old_state = record.health_state
   697→            record.update_status()
   698→            # Si no hay heartbeat reciente, marcar como DEAD.
   699→            try:
   700→                last_hb = datetime.fromisoformat(record.last_heartbeat).timestamp()
   701→                if now_epoch - last_hb > self._heartbeat_timeout:
   702→                    if record.health_state != HealthState.DEAD:
   703→                        record.health_state = HealthState.DEAD
   704→            except (ValueError, OSError):
   705→                pass
   706→            if record.health_state != old_state:
   707→                changes += 1
   708→                # Emitir evento de cambio de health (best-effort).
   709→                if self._bus is not None:
   710→                    try:
   711→                        asyncio.create_task(self._emit(
   712→                            "agent.health.changed",
   713→                            {
   714→                                "agent_id": record.id,
   715→                                "name": record.name,
   716→                                "old_state": old_state.value,
   717→                                "new_state": record.health_state.value,
   718→                            },
   719→                        ))
   720→                    except Exception:  # noqa: BLE001
   721→                        pass
   722→        self._metrics["heartbeat_checks"] += 1
   723→        self._metrics["health_changes"] += changes
   724→        return changes
   725→
   726→    def health(self) -> dict[str, Any]:
   727→        """Health check de todos los agentes."""
   728→        records = self.get_all()
   729→        summary: dict[str, int] = {s.value: 0 for s in HealthState}
   730→        for r in records:
   731→            summary[r.health_state.value] += 1
   732→        return {
   733→            "total": len(records),
   734→            "summary": summary,
   735→            "healthy": summary.get(HealthState.ALIVE.value, 0)
   736→                       + summary.get(HealthState.IDLE.value, 0),
   737→            "degraded": summary.get(HealthState.BUSY.value, 0)
   738→                        + summary.get(HealthState.ERROR.value, 0),
   739→            "dead": summary.get(HealthState.DEAD.value, 0),
   740→            "agents": {r.name: r.health_state.value for r in records},
   741→        }
   742→
   743→    async def start_heartbeat_loop(self, interval: float = 10.0) -> None:
   744→        """Arranca un background task que actualiza health periódicamente."""
   745→        if self._heartbeat_task is not None:
   746→            return
   747→
   748→        async def _loop() -> None:
   749→            while True:
   750→                try:
   751→                    self.update_health()
   752→                except Exception:  # noqa: BLE001
   753→                    logger.exception("Heartbeat loop failed")
   754→                await asyncio.sleep(interval)
   755→
   756→        self._heartbeat_task = asyncio.create_task(_loop())
   757→        logger.info("AgentRegistry heartbeat loop started (interval=%.1fs)", interval)
   758→
   759→    async def stop_heartbeat_loop(self) -> None:
   760→        """Detiene el heartbeat loop."""
   761→        if self._heartbeat_task is not None:
   762→            self._heartbeat_task.cancel()
   763→            try:
   764→                await self._heartbeat_task
   765→            except asyncio.CancelledError:
   766→                pass
   767→            self._heartbeat_task = None
   768→            logger.info("AgentRegistry heartbeat loop stopped")

   869→    def list_manifests(self) -> list["AgentManifest"]:
   870→        """Devuelve el manifest de cada agente registrado (Sprint 2.3).
   871→
   872→        Si un agente no tiene manifest explícito, se construye uno
   873→        sintético a partir de su AgentRecord.
   874→        """
   875→        from .manifest import AgentManifest
   876→        manifests: list[AgentManifest] = []
   877→        for record in self.get_all():
   878→            # Si el agente expone un método manifest(), usarlo.
   879→            agent = record.agent
   880→            if hasattr(agent, "manifest") and callable(agent.manifest):
   881→                try:
   882→                    m = agent.manifest()
   883→                    # Duck-typing: aceptar cualquier objeto con los campos
   884→                    # clave (id, name, version, capabilities, priority) para
   885→                    # ser robusto frente a re-imports del módulo AgentManifest.
   886→                    if (
   887→                        hasattr(m, "id")
   888→                        and hasattr(m, "name")
   889→                        and hasattr(m, "version")
   890→                        and hasattr(m, "capabilities")
   891→                        and hasattr(m, "priority")
   892→                    ):
   893→                        manifests.append(m)  # type: ignore[arg-type]
   894→                        continue
   895→                except Exception:  # noqa: BLE001
   896→                    pass
   897→            # Sintetizar desde el record.
   898→            manifests.append(AgentManifest(
   899→                id=record.name.lower().replace(" ", "_"),
   900→                name=record.name,
   901→                version=record.version,
   902→                description=record.description,
   903→                capabilities=list(record.provided_capabilities),
   904→                priority=record.priority,
   905→                metadata=dict(record.metadata),
   906→            ))
   907→        return manifests

   932→    async def _emit(self, event_type: str, payload: dict[str, Any]) -> None:
   933→        """Emite un evento al EventBus (best-effort)."""
   934→        if self._bus is None:
   935→            return
   936→        try:
   937→            await self._bus.publish(event_type, source="agent_registry", payload=payload)
   938→        except Exception:  # noqa: BLE001
   939→            logger.exception("Failed to emit event '%s'", event_type)

   941→
   942→def _pattern_matches_cap(pattern: str, cap_name: str) -> bool:
   943→    """True si `pattern` (con wildcards) matchea `cap_name`."""
   944→    if pattern == "*":
   945→        return True
   946→    if pattern == cap_name:
   947→        return True
   948→    if pattern.endswith(".*"):
   949→        prefix = pattern[:-2]
   950→        return cap_name == prefix or cap_name.startswith(prefix + ".")
   951→    # Match jerárquico: pattern="code" matches cap_name="code.generate".
   952→    if cap_name.startswith(pattern + "."):
   953→        return True
   954→    return False
